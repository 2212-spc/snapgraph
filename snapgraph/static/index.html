function patchTrustHeader() { document.querySelectorAll('.trust-center-header').forEach((header) => { if (header.dataset.trustHeaderPatched === 'true') { return; } const eyebrow = header.querySelector('.eyebrow'); const title = header.querySelector('h1'); const paragraphs = header.querySelectorAll('p:not(.eyebrow)'); if (eyebrow) { eyebrow.textContent = '信任'; } if (title) { title.textContent = '检查 AI 猜测有没有冒充你的想法'; } if (paragraphs[0]) { paragraphs[0].textContent = '把 AI 推断、证据路径、open loop 和审查历史放在一个可操作的工作台里。'; } header.dataset.trustHeaderPatched = 'true'; }); } let scheduled = false; const observer = new MutationObserver(() => { if (scheduled) { return; } scheduled = true; requestAnimationFrame(() => { scheduled = false; patchTrustHeader(); }); }); observer.observe(document.documentElement, { childList: true, subtree: true }); if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', patchTrustHeader, { once: true }); } else { patchTrustHeader(); } })();
